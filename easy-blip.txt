### relatively easy options for blocking a whole bunch of attackers who have been constantly hitting your firewall-less linux box over SSH
### ill put this in a script altogether later, but for now this is easy enough
### it could also be easily daemonized so you dont have to worry about it
### probably could be nicely integrated into fail2ban as well

# regex for ip4s as "block list I P" stored as variable -- in bashrc for me
$ echo "$BLIP"
[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}

# put aggressive ips into a file
$ grep -E -o "$BLIP" $(journalctl -u sshd) > /home/k/blips.txt
# optionally, it might be wise to remove your own ip here (i keep forgetting to do this) -->
# $ grep -E -o "$BLIP" $(journalctl -u sshd) | grep -v my.ip.addr.here > /home/k/blips.txt

# cut the ips out one by one, sort the list, dedupe the list, store in new file
$ for i in ~/journal.txt; do grep -oP "$BLIP" "$i" | sort | uniq > blipsonly.txt ; done

# remove my IP (1.1.1.1 for example) from the file
$ grep -v 1\.\1\.1\.1 blipsonly.txt >> blipsonlynohome.txt

# make the iptables stuff
$ sudo ipset create blocklist hash:ip hashsize 4096
$ sudo iptables -I INPUT -m set --match-set blocklist src -j DROP
$ sudo iptables -I FORWARD -m set --match-set blocklist src -j DROP

### script to finish it off, chmod 755 and execute with sudo or root (can be run just as a command with sudo)
$ cat blbl.sh
#!/bin/bash

while read ip; do ipset add blocklist $ip; done < /home/k/blipsonlynohome.txt
$ sudo ./blbl.sh

# check that it worked with
$ sudo ipset list

### rinse and repeat for continued success until your iptables fill up
